cmake_minimum_required(VERSION 3.24)
project(5050-censor-filter VERSION 0.2.1 LANGUAGES CXX)

add_library(5050-censor-filter MODULE src/filter.cpp src/video.cpp src/indicator.cpp)
add_library(obs-sdk INTERFACE)
target_link_libraries(5050-censor-filter PRIVATE obs-sdk user32 gdi32 comctl32)
target_compile_features(5050-censor-filter PRIVATE cxx_std_17)
target_compile_definitions(5050-censor-filter PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(5050-censor-filter PRIVATE /utf-8 /W4 /permissive-)
endif()

# Prefer an installed OBS development package. An unpacked official source
# release and matching Windows import library can also be used offline.
set(OBS_SOURCE_DIR "" CACHE PATH "OBS source tree containing libobs")
set(OBS_IMPORT_LIB "" CACHE FILEPATH "Windows import library for obs.dll")
if(OBS_SOURCE_DIR AND OBS_IMPORT_LIB)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/obsconfig.h"
    "#pragma once\n#define OBS_RELEASE_CANDIDATE 0\n#define OBS_BETA 0\n#define OBS_DATA_PATH \"\"\n#define OBS_INSTALL_PREFIX \"\"\n")
  target_include_directories(obs-sdk SYSTEM INTERFACE
    "${OBS_SOURCE_DIR}/libobs" "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(obs-sdk INTERFACE "${OBS_IMPORT_LIB}")
  get_filename_component(_obs_lib_dir "${OBS_IMPORT_LIB}" DIRECTORY)
  find_library(OBS_FRONTEND_IMPORT_LIB NAMES obs-frontend-api HINTS "${_obs_lib_dir}" REQUIRED)
  target_include_directories(5050-censor-filter SYSTEM PRIVATE "${OBS_SOURCE_DIR}/frontend/api")
  target_link_libraries(5050-censor-filter PRIVATE "${OBS_FRONTEND_IMPORT_LIB}")
else()
  find_package(libobs CONFIG REQUIRED)
  find_package(obs-frontend-api CONFIG REQUIRED)
  target_link_libraries(obs-sdk INTERFACE OBS::libobs)
  target_link_libraries(5050-censor-filter PRIVATE OBS::obs-frontend-api)
endif()
set_target_properties(5050-censor-filter PROPERTIES PREFIX "")
install(TARGETS 5050-censor-filter DESTINATION 5050-censor-filter)
install(DIRECTORY data/ DESTINATION 5050-censor-filter/data)
install(FILES README.md DESTINATION 5050-censor-filter)

option(BUILD_CAPTURE_TEST "Build the standalone libobs A/V integration test" OFF)
if(BUILD_CAPTURE_TEST)
  add_executable(capture-test tests/capture.cpp)
  target_compile_features(capture-test PRIVATE cxx_std_17)
  target_link_libraries(capture-test PRIVATE obs-sdk)
  if(WIN32)
    add_executable(capture-fixture WIN32 tests/window.cpp)
    target_link_libraries(capture-fixture PRIVATE user32 gdi32 winmm)
  endif()
endif()
